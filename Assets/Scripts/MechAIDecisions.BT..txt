tree("Root")
	repeat
		mute
			fallback
				tree "EmergencyFlee"
				tree "Attacking"
				tree "Pursuing"
				tree "TacticalRetreat"
				tree "Roaming"
tree("EmergencyFlee")
	while
		sequence
			HasAttackTarget()
			StatusCheck()
			CriticalHealth()
	repeat
		Flee()
tree ("Attacking")
	while
		sequence
			HasAttackTarget()
			TargetLOS()
			not StatusCheck()
		repeat
			Attack()
tree ("Pursuing")
	while
		sequence
			HasAttackTarget()
			not TargetLOS()
			not StatusCheck()
		repeat
			Pursue()
tree("TacticalRetreat")
	while
		sequence
			HasAttackTarget()
			StatusCheck()
			not CriticalHealth()
	repeat
		TacticalRetreat()
tree ("Roaming")
	while not HasAttackTarget()
		repeat
			Roam()
